void Check(bool condition, string name)
{
    if (!condition) throw new Exception("FALLO: " + name);
    Console.WriteLine("OK: " + name);
    checks++;
}
void Rejected(Action action, string name, string expectedNumber = "23505")
{
    try { action(); }
    catch (PostgresException ex) when (ex.SqlState == expectedNumber) { Check(true, name); return; }
    throw new Exception("FALLO: se acepto una operacion invalida: " + name);
}
